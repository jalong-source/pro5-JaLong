Máy Tính Giảm Giá (Discount Calculator)

Nhập số tiền hóa đơn mua sắm. Nếu hóa đơn lớn hơn 500.000 VNĐ, hệ thống tự động giảm giá 10%, ngược lại giữ nguyên mức thanh toán.

💡 Ưu đãi: Hóa đơn > 500.000 VNĐ được tự động giảm ngay 10%.

Hóa đơn gốc: 650,000 VNĐ
Mức giảm giá: 10% (-65,000 VNĐ)
Tổng số tiền phải trả: 585,000 VNĐ

Mã nguồn JavaScript mẫu

function tinhGiamGia() {
  let tongTien = parseFloat(prompt("Nhập số tiền hóa đơn (VNĐ):", "650000"));

  if (isNaN(tongTien) || tongTien < 0) {
    alert("Số tiền không hợp lệ!");
    return;
  }

  let giamGia = 0;
  // Kiểm tra nếu hóa đơn > 500.000 VNĐ thì giảm 10%
  if (tongTien > 500000) {
    giamGia = tongTien * 0.10;
  }

  let thanhToan = tongTien - giamGia;
  let msg = `Hóa đơn: ${tongTien.toLocaleString('vi-VN')} đ\n` +
            `Giảm giá: ${giamGia.toLocaleString('vi-VN')} đ\n` +
            `Số tiền phải trả: ${thanhToan.toLocaleString('vi-VN')} đ`;

  alert(msg);
  console.log(msg);
}
