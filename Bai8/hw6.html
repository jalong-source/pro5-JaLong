Phân Loại Tam Giác (Kiểm Tra Tam Giác Đều)

Nhập vào độ dài 3 cạnh a, b, c của một tam giác. Xác thực điều kiện hình học tam giác và kiểm tra xem đó có phải là tam giác đều (3 cạnh bằng nhau) hay không.

Bộ ba cạnh: (a=6, b=6, c=6)
Kết luận phân loại: TAM GIÁC ĐỀU 🔺

Mã nguồn JavaScript mẫu

function kiemTraTamGiacDeu() {
  let a = parseFloat(prompt("Nhập cạnh a:", "6"));
  let b = parseFloat(prompt("Nhập cạnh b:", "6"));
  let c = parseFloat(prompt("Nhập cạnh c:", "6"));

  // 1. Kiểm tra cạnh dương
  if (isNaN(a) || isNaN(b) || isNaN(c) || a <= 0 || b <= 0 || c <= 0) {
    alert("Độ dài các cạnh phải là số dương hợp lệ!");
    return;
  }

  // 2. Bất đẳng thức tam giác
  if (a + b <= c || a + c <= b || b + c <= a) {
    alert("Ba cạnh này KHÔNG thể tạo thành một hình tam giác!");
    return;
  }

  // 3. Điều kiện tam giác đều
  if (a === b && b === c) {
    alert(`Ba cạnh (${a}, ${b}, ${c}) tạo thành: TAM GIÁC ĐỀU! 🔺`);
    console.log("Đây là tam giác đều.");
  } else {
    alert(`Ba cạnh (${a}, ${b}, ${c}) KHÔNG PHẢI là tam giác đều.`);
    console.log("Không phải tam giác đều.");
  }
}
