Bài 2: "Logic Builder" — Máy tính kiểm tra xếp loại điểm

Sử dụng prompt() để nhận điểm từ người dùng, cấu trúc điều kiện if - else if - else để xếp loại học tập và thông báo qua alert().

Bấm nút bên dưới để mở hộp thoại nhập điểm:

1. Tiêu chí xếp loại học tập

  • Điểm > 8.0: Xếp loại Giỏi (Xuất sắc)
  • Điểm > 6.5: Xếp loại Khá
  • Điểm >= 5.0: Xếp loại Trung bình (Đạt)
  • Điểm < 5.0: Xếp loại Yếu (Chưa đạt)

2. Đoạn mã JavaScript đầy đủ

function kiemTraDiem() {
  // 1. Nhận điểm từ prompt
  let input = prompt("Nhập điểm số của bạn (từ 0 đến 10):", "8.5");

  if (input === null || input.trim() === "") {
    alert("Bạn đã hủy thao tác hoặc chưa nhập điểm!");
    return;
  }

  let score = parseFloat(input);

  // 2. Kiểm tra dữ liệu hợp lệ
  if (isNaN(score) || score < 0 || score > 10) {
    alert("Điểm không hợp lệ! Vui lòng nhập số từ 0 đến 10.");
    return;
  }

  // 3. Xử lý phân loại bằng if - else if - else
  let rank = "";
  let feedback = "";

  if (score > 8.0) {
    rank = "Giỏi 🌟";
    feedback = "Chúc mừng bạn đạt kết quả xuất sắc!";
  } else if (score > 6.5) {
    rank = "Khá 👍";
    feedback = "Làm tốt lắm, tiếp tục phát huy nhé!";
  } else if (score >= 5.0) {
    rank = "Trung bình 👏";
    feedback = "Chúc mừng bạn đã đạt!";
  } else {
    rank = "Yếu 📚";
    feedback = "Rất tiếc, hãy cố gắng hơn ở lần sau!";
  }

  // 4. Hiển thị thông báo
  let message = `Điểm số của bạn: ${score}\nXếp loại: ${rank}\nĐánh giá: ${feedback}`;
  alert(message);
  console.log(message);
}
