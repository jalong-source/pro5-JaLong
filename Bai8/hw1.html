Bài Tập Về Nhà 01 · Tỉ Giá 25.000

Chuyển Đổi Tiền Tệ (USD ➔ VND)

Nhập số lượng USD cần đổi và tính ra tổng tiền Việt Nam Đồng (VND) tương ứng dựa trên tỉ giá cố định 25.000 VNĐ / USD.

Số tiền USD: 10 USD
Tỉ giá áp dụng: 25,000 VNĐ / USD
Thành tiền: 250,000 VNĐ

Mã nguồn JavaScript mẫu

function doiTienUSD() {
  const TI_GIA = 25000; // 1 USD = 25.000 VNĐ
  let usd = parseFloat(prompt("Nhập số tiền USD cần đổi:", "10"));

  if (isNaN(usd) || usd < 0) {
    alert("Số tiền USD không hợp lệ!");
    return;
  }

  let vnd = usd * TI_GIA;
  let ketQua = `${usd} USD = ${vnd.toLocaleString('vi-VN')} VNĐ`;

  // Xuất ra console và alert
  console.log(ketQua);
  alert(ketQua);
}
