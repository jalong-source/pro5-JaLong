Tính Chỉ Số Khối Cơ Thể (BMI)

Nhập chiều cao (m) và cân nặng (kg) để tính toán chỉ số BMI = cân nặng / (chiều cao * chiều cao) và phân loại tình trạng thể lực.

Chỉ số BMI của bạn: 21.97
Tình trạng thể lực: Bình thường (Cân đối)
Thang chuẩn: <18.5 (Gầy) | 18.5 - 24.9 (Cân đối) | 25 - 29.9 (Thừa cân) | ≥30 (Béo phì)

Mã nguồn JavaScript mẫu

function tinhBMI() {
  let chieuCao = parseFloat(prompt("Nhập chiều cao của bạn (m - ví dụ: 1.72):", "1.72"));
  let canNang = parseFloat(prompt("Nhập cân nặng của bạn (kg - ví dụ: 65):", "65"));

  if (isNaN(chieuCao) || isNaN(canNang) || chieuCao <= 0 || canNang <= 0) {
    alert("Chiều cao và cân nặng phải là số dương hợp lệ!");
    return;
  }

  let bmi = canNang / (chieuCao * chieuCao);
  let status = "";

  if (bmi < 18.5) status = "Gầy (Thiếu cân)";
  else if (bmi < 24.9) status = "Bình thường (Cân đối)";
  else if (bmi < 29.9) status = "Tiền béo phì (Thừa cân)";
  else status = "Béo phì";

  let msg = `Chỉ số BMI: ${bmi.toFixed(2)} - Đánh giá: ${status}`;
  alert(msg);
  console.log(msg);
}
